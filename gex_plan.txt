## OVERVIEW
================================================================================

GEX (Gamma Exposure) measures how sensitive market maker delta hedging is to
price changes. When integrated with pattern-based trading:

- Positive GEX = Mean reversion environment = GOOD for reversal trades
- Negative GEX = Trend amplification = RISKY for reversal trades

Research shows trades informed by options flow data have 15% higher success rate.

# File: algotrade/gex_service.py
# pip install requests pandas numpy

import requests
import pandas as pd
import numpy as np
import time
import logging

logger = logging.getLogger(__name__)


class TradierGEX:
    """Free GEX calculator using Tradier sandbox API."""

    def __init__(self, api_token: str, sandbox: bool = True):
        self.api_token = api_token
        self.base_url = "https://sandbox.tradier.com" if sandbox else "https://api.tradier.com"
        self.headers = {
            "Authorization": f"Bearer {api_token}",
            "Accept": "application/json"
        }

    def get_options_chain(self, symbol: str, expiration: str = None) -> pd.DataFrame:
        """Fetch options chain with Greeks from Tradier."""
        # Get expirations if not specified
        if not expiration:
            expiration = self._get_nearest_expiration(symbol)

        url = f"{self.base_url}/v1/markets/options/chains"
        params = {
            "symbol": symbol,
            "expiration": expiration,
            "greeks": "true"  # Include Greeks!
        }

        response = requests.get(url, headers=self.headers, params=params)
        data = response.json()

        if "options" in data and data["options"]:
            return pd.DataFrame(data["options"]["option"])
        return pd.DataFrame()

    def _get_nearest_expiration(self, symbol: str) -> str:
        """Get nearest expiration (0DTE if available)."""
        url = f"{self.base_url}/v1/markets/options/expirations"
        params = {"symbol": symbol}
        response = requests.get(url, headers=self.headers, params=params)
        data = response.json()

        if "expirations" in data and data["expirations"]:
            return data["expirations"]["date"][0]  # First = nearest
        return None

    def calculate_gex(self, symbol: str, spot_price: float = None) -> dict:
        """
        Calculate Gamma Exposure (GEX) for a symbol.

        Assumes dealers are:
        - LONG calls (positive gamma)
        - SHORT puts (negative gamma, flipped to positive for calc)
        """
        chain = self.get_options_chain(symbol)

        if chain.empty:
            return {"error": "No options data"}

        # Get spot price if not provided
        if not spot_price:
            spot_price = self._get_spot_price(symbol)

        contract_size = 100

        # Calculate GEX per strike
        # Call GEX = gamma * OI * contract_size * spot^2 * 0.01
        # Put GEX = -gamma * OI * contract_size * spot^2 * 0.01 (dealers short puts)

        calls = chain[chain["option_type"] == "call"].copy()
        puts = chain[chain["option_type"] == "put"].copy()

        # Extract gamma from Greeks
        calls["gamma"] = calls["greeks"].apply(
            lambda x: x.get("gamma", 0) if isinstance(x, dict) else 0
        )
        puts["gamma"] = puts["greeks"].apply(
            lambda x: x.get("gamma", 0) if isinstance(x, dict) else 0
        )

        calls["gex"] = (
            calls["gamma"] * calls["open_interest"] *
            contract_size * spot_price**2 * 0.01
        )
        puts["gex"] = (
            -puts["gamma"] * puts["open_interest"] *
            contract_size * spot_price**2 * 0.01
        )

        # Aggregate by strike
        call_gex = calls.groupby("strike")["gex"].sum()
        put_gex = puts.groupby("strike")["gex"].sum()

        total_gex = call_gex.add(put_gex, fill_value=0)
        net_gex = total_gex.sum()

        # Find key levels
        max_gamma_strike = total_gex.idxmax() if not total_gex.empty else None

        # Find gamma flip (where cumulative GEX crosses zero)
        cumulative = total_gex.sort_index().cumsum()
        flip_candidates = cumulative[cumulative.shift(1, fill_value=0) * cumulative < 0]
        gamma_flip = flip_candidates.index[0] if len(flip_candidates) > 0 else None

        return {
            "symbol": symbol,
            "spot_price": spot_price,
            "net_gex": net_gex,
            "net_gex_billions": net_gex / 1e9,
            "gamma_flip": gamma_flip,  # HVL level
            "max_gamma_strike": max_gamma_strike,
            "gex_by_strike": total_gex.to_dict(),
            "gex_positive": net_gex > 0,  # True = mean reversion environment
            "interpretation": "Mean reversion likely" if net_gex > 0 else "Trend amplification likely"
        }

    def _get_spot_price(self, symbol: str) -> float:
        """Get current spot price."""
        url = f"{self.base_url}/v1/markets/quotes"
        params = {"symbols": symbol}
        response = requests.get(url, headers=self.headers, params=params)
        data = response.json()

        if "quotes" in data and data["quotes"]:
            return data["quotes"]["quote"]["last"]
        return 0


class GEXChecker:
    """GEX environment checker for trade confirmation."""

    def __init__(self, tradier_token: str):
        self.gex_client = TradierGEX(tradier_token, sandbox=True)
        self._cached_gex = None
        self._cache_time = None

    def get_gex_environment(self, symbol: str = "SPY") -> dict:
        """Get current GEX environment. Cache for 5 minutes."""
        if self._cached_gex and self._cache_time:
            if time.time() - self._cache_time < 300:  # 5 min cache
                return self._cached_gex

        try:
            self._cached_gex = self.gex_client.calculate_gex(symbol)
            self._cache_time = time.time()
            return self._cached_gex
        except Exception as e:
            logger.error(f"GEX fetch failed: {e}")
            return {"gex_positive": None, "error": str(e)}

    def should_trade_reversal(self, spy_price: float) -> tuple:
        """
        Check if GEX environment supports reversal trade.

        Returns: (should_trade: bool, reason: str)
        """
        gex = self.get_gex_environment()

        if "error" in gex:
            return True, "GEX unavailable, proceeding with pattern"

        gamma_flip = gex.get("gamma_flip")
        is_positive = gex.get("gex_positive")

        # Positive GEX = mean reversion environment = good for reversals
        if is_positive:
            return True, f"Positive GEX (${gex['net_gex_billions']:.1f}B) - reversal supported"

        # Negative GEX but above gamma flip = still ok
        if gamma_flip and spy_price > gamma_flip:
            return True, f"Above gamma flip ({gamma_flip}) - reversal ok"

        # Negative GEX below gamma flip = risky for reversals
        return False, f"Negative GEX below flip ({gamma_flip}) - skip reversal"


# Standalone test
if __name__ == "__main__":

    print(f"  Max Gamma Strike: {result['max_gamma_strike']}")
    print(f"  Environment: {result['interpretation']}")

## GEX INTERPRETATION GUIDE
================================================================================

### Key Levels:

1. HVL (High Volatility Level / Gamma Flip)
   - Above HVL: Positive gamma, low volatility, mean reversion
   - Below HVL: Negative gamma, high volatility, trend continuation

2. Gamma Walls
   - Call Wall = Resistance (dealers sell into rallies)
   - Put Wall = Support (dealers buy into dips)

### Trading Rules:

| GEX State      | SPY Position    | Reversal Trade? |
|----------------|-----------------|-----------------|
| Positive       | Any             | YES - supported |
| Negative       | Above flip      | YES - cautious  |
| Negative       | Below flip      | NO - skip trade |

### Log Output Example:

```

[GEX Check] Above gamma flip (585.00) - reversal ok
[GEX Check] Negative GEX below flip (585.00) - skip reversal
```
